cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(autofinance_setup_bridge LANGUAGES CXX)
# Inno 6.5.4 is a Win32 process. This private in-process adapter is Win32;
# the application and its payload remain AMD64. No x86 CRT is installed.
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build the Inno 6 bridge with -A Win32")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("../../../windows/maintenance" maintenance)
add_library(installer_files STATIC installer_files.cpp)
target_compile_features(installer_files PUBLIC cxx_std_17)
target_compile_definitions(installer_files PUBLIC UNICODE _UNICODE NOMINMAX)
target_compile_options(installer_files PUBLIC /W4 /WX /EHsc /utf-8)
target_include_directories(installer_files PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(installer_files PUBLIC bcrypt)
add_library(autofinance_setup_bridge SHARED setup_bridge.cpp setup_bridge.def)
target_link_libraries(autofinance_setup_bridge PRIVATE installer_files autofinance_maintenance)
enable_testing()
add_executable(installer_files_tests tests/installer_files_test.cpp)
target_link_libraries(installer_files_tests PRIVATE installer_files)
add_test(NAME installer_files COMMAND installer_files_tests)
add_executable(bridge_abi_tests tests/bridge_abi_test.cpp)
target_compile_features(bridge_abi_tests PRIVATE cxx_std_17)
add_dependencies(bridge_abi_tests autofinance_setup_bridge)
add_test(NAME bridge_abi COMMAND bridge_abi_tests "$<TARGET_FILE:autofinance_setup_bridge>")

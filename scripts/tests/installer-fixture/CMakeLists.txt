cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(autofinance_installer_fixture LANGUAGES CXX RC)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Fixture must be x64")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(revision 1 2)
  add_executable(fixture${revision} WIN32 main.cpp identity.rc)
  target_compile_definitions(fixture${revision} PRIVATE FIXTURE_REVISION=${revision})
  set_target_properties(fixture${revision} PROPERTIES OUTPUT_NAME myautofinance RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/v${revision}")
endforeach()
add_executable(verify_fixture verify.cpp)
target_link_libraries(verify_fixture PRIVATE version)

cmake_minimum_required(VERSION 3.14)
project(autofinance_maintenance LANGUAGES CXX)
add_library(autofinance_maintenance STATIC session_maintenance.cpp windows_session_store.cpp)
target_compile_features(autofinance_maintenance PUBLIC cxx_std_17)
target_compile_definitions(autofinance_maintenance PUBLIC UNICODE _UNICODE NOMINMAX)
target_link_libraries(autofinance_maintenance PUBLIC advapi32 bcrypt shell32 ole32 version)
target_include_directories(autofinance_maintenance PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(autofinance_maintenance PUBLIC /W4 /WX /EHsc /utf-8)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(maintenance_tests tests/session_maintenance_test.cpp)
  target_link_libraries(maintenance_tests PRIVATE autofinance_maintenance)
  add_test(NAME maintenance_state_machine COMMAND maintenance_tests)
  add_executable(maintenance_native_tests tests/windows_session_store_test.cpp)
  target_link_libraries(maintenance_native_tests PRIVATE autofinance_maintenance)
  # Intentionally NOT in ctest: requires a disposable Windows user and fixture EXEs.
endif()
